# SSDF Okta + Wazuh Connectors Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Scaffold the Python tooling (`uv`, `pytest`, `ruff`) under `py/` and implement two agentless pull-connectors: an Okta connector that polls the System Log API (cursor-based) and a Wazuh connector that polls the Wazuh indexer REST API. Each connector publishes raw vendor JSON to its topic (`raw.okta`, `raw.wazuh`) and also normalizes records directly into canonical `Event` JSON published to `events.normalized`. Python connectors own their own normalization (the Rust normalizer handles SRX/PAN-OS only). Okta produces `auth_event`, `policy_decision_event`, and `alert_event`; Wazuh produces `alert_event`.

**Architecture:** A thin client class (mockable) wraps each vendor API and is injected into a polling loop. Mapping functions are pure (input: vendor dict → output: canonical `Event` dict) and fully unit-tested with fixture JSON. A shared `ssdf_events.py` module defines typed dataclasses/pydantic models that serialize to the canonical shape. A shared `bus.py` module wraps `aiokafka` (async) with a clean `publish(topic, key, value)` interface. Cursor/offset state is persisted to a local JSON file so restarts resume from the last seen position.

**Tech Stack:** Python 3.12, `uv` (workspace + lockfile), `pydantic` v2, `aiokafka`, `httpx` (async HTTP client), `python-ulid`, `pytest` + `pytest-asyncio`, `ruff` (lint + format). No external SaaS dependencies; Redpanda is the only required backing service (already running via docker-compose from Plan 1).

**Spec:** `docs/superpowers/specs/2026-06-05-ssdf-data-fabric-design.md` (§3 Ontology, §4 Data Plane — Okta/Wazuh ingest, §2 build order).

```python
"""
Canonical SSDF Event model — Python mirror of the Rust ssdf-ontology crate's Event JSON.

Shape contract (flat JSON object):
  event_id       str          "evt_<ULID>"
  tenant_id      str          default "t_main"
  ts             str          RFC3339 UTC  e.g. "2026-06-05T14:09:40Z"
  source_type    str          "okta" | "wazuh" | "srx" | "panos"
  source_instance str
  severity       SeverityEnum info|low|medium|high|critical
  event_type     str          snake_case discriminator (set by each payload subclass)
  refs           dict | None  {identity_id?, asset_id?, app_id?, policy_id?, session_id?}
  ext            dict | None  {okta: {...}} or {wazuh: {...}} for non-canonical fields

Plus payload fields promoted to top level (flattened).
"""
from __future__ import annotations

from enum import Enum
from typing import Any

from pydantic import BaseModel, Field, model_serializer
from ulid import ULID


class SeverityEnum(str, Enum):
    info = "info"
    low = "low"
    medium = "medium"
    high = "high"
    critical = "critical"


def _new_event_id() -> str:
    return f"evt_{ULID()}"


class EventRefs(BaseModel):
    """Optional cross-entity reference IDs resolved by the Entity Resolution service."""

    identity_id: str | None = None
    asset_id: str | None = None
    app_id: str | None = None
    policy_id: str | None = None
    session_id: str | None = None

    model_config = {"extra": "forbid"}

    def as_dict(self) -> dict[str, Any]:
        return {k: v for k, v in self.model_dump().items() if v is not None}


class _EventBase(BaseModel):
    """Shared envelope — not instantiated directly."""

    event_id: str = Field(default_factory=_new_event_id)
    tenant_id: str = "t_main"
    ts: str
    source_type: str
    source_instance: str
    severity: SeverityEnum
    refs: EventRefs | None = None
    ext: dict[str, Any] | None = None

    # Subclasses declare event_type as a ClassVar or literal field.
    event_type: str

    model_config = {"extra": "forbid"}

    @model_serializer(mode="wrap")
    def _flat_serialize(self, handler: Any) -> dict[str, Any]:
        """Produce a flat dict: envelope + payload fields at the same level."""
        base = handler(self)
        # Promote refs contents inline, drop None refs key
        refs_obj = base.pop("refs", None)
        ext_obj = base.pop("ext", None)
        if refs_obj:
            base["refs"] = {k: v for k, v in refs_obj.items() if v is not None}
        if ext_obj:
            base["ext"] = ext_obj
        return base


class AuthEventModel(_EventBase):
    event_type: str = "auth_event"
    source_type: str = "okta"
    actor: str
    outcome: str
    mfa: bool
    auth_method: str
    src_ip: str
    geo: str | None = None
    risk: str | None = None


class PolicyDecisionEventModel(_EventBase):
    event_type: str = "policy_decision_event"
    source_type: str = "okta"
    policy_ref: str
    decision: str
    reason: str | None = None
    matched_on: str | None = None


class AlertEventModel(_EventBase):
    event_type: str = "alert_event"
    rule_id: str
    title: str
    category: str
    confidence: float | None = None
    affected_ip: str | None = None
    affected_user: str | None = None


def build_auth_event(
    *,
    tenant_id: str,
    source_instance: str,
    ts: str,
    severity: str,
    actor: str,
    outcome: str,
    mfa: bool,
    auth_method: str,
    src_ip: str,
    geo: str | None,
    risk: str | None,
    refs: dict[str, str] | None,
    ext_okta: dict[str, Any] | None,
) -> AuthEventModel:
    """Build a canonical AuthEvent. source_type is always 'okta'."""
    return AuthEventModel(
        tenant_id=tenant_id,
        source_instance=source_instance,
        ts=ts,
        severity=SeverityEnum(severity),
        actor=actor,
        outcome=outcome,
        mfa=mfa,
        auth_method=auth_method,
        src_ip=src_ip,
        geo=geo,
        risk=risk,
        refs=EventRefs(**refs) if refs else None,
        ext={"okta": ext_okta} if ext_okta else None,
    )


def build_policy_decision_event(
    *,
    tenant_id: str,
    source_instance: str,
    ts: str,
    severity: str,
    policy_ref: str,
    decision: str,
    reason: str | None,
    matched_on: str | None,
    refs: dict[str, str] | None,
    ext_okta: dict[str, Any] | None,
) -> PolicyDecisionEventModel:
    """Build a canonical PolicyDecisionEvent. source_type is always 'okta'."""
    return PolicyDecisionEventModel(
        tenant_id=tenant_id,
        source_instance=source_instance,
        ts=ts,
        severity=SeverityEnum(severity),
        policy_ref=policy_ref,
        decision=decision,
        reason=reason,
        matched_on=matched_on,
        refs=EventRefs(**refs) if refs else None,
        ext={"okta": ext_okta} if ext_okta else None,
    )


def build_alert_event(
    *,
    tenant_id: str,
    source_type: str,
    source_instance: str,
    ts: str,
    severity: str,
    rule_id: str,
    title: str,
    category: str,
    confidence: float | None,
    affected_ip: str | None,
    affected_user: str | None,
    refs: dict[str, str] | None,
    ext_vendor: dict[str, Any] | None,
    ext_vendor_key: str,
) -> AlertEventModel:
    """Build a canonical AlertEvent for any source (okta, wazuh, srx, panos)."""
    return AlertEventModel(
        tenant_id=tenant_id,
        source_type=source_type,
        source_instance=source_instance,
        ts=ts,
        severity=SeverityEnum(severity),
        rule_id=rule_id,
        title=title,
        category=category,
        confidence=confidence,
        affected_ip=affected_ip,
        affected_user=affected_user,
```

```python
"""
JSON-file-backed cursor for connector poll position persistence.

Each cursor instance manages a single named key within a shared JSON state file,
so multiple connectors can share one file or use separate files per connector.
"""
from __future__ import annotations

import json
from pathlib import Path


class Cursor:
    """Persist and retrieve a single string cursor value keyed within a JSON file."""

    def __init__(self, state_file: Path, key: str) -> None:
        self._path = state_file
        self._key = key

    def get(self) -> str | None:
        """Return the stored cursor, or None if the file/key does not exist."""
        if not self._path.exists():
            return None
        try:
            data = json.loads(self._path.read_text())
            return data.get(self._key)
        except (json.JSONDecodeError, OSError):
            return None

    def set(self, value: str) -> None:
        """Persist the cursor value. Atomic: reads, merges, writes."""
        existing: dict[str, str] = {}
        if self._path.exists():
            try:
                existing = json.loads(self._path.read_text())
            except (json.JSONDecodeError, OSError):
                existing = {}
        existing[self._key] = value
        # Write atomically via a temp sibling then rename
        tmp = self._path.with_suffix(".tmp")
        tmp.write_text(json.dumps(existing, indent=2))
        tmp.replace(self._path)
```

```python
"""
Async message bus wrapper — publishes raw vendor bytes and canonical Event JSON to Redpanda.

The producer is injected at construction time so the bus is unit-testable without a live broker.
For production, pass an `aiokafka.AIOKafkaProducer` instance (already started).
"""
from __future__ import annotations

from typing import Any, Protocol, runtime_checkable

from connectors.common.ssdf_events import _EventBase


@runtime_checkable
class _Producer(Protocol):
    async def send_and_wait(self, topic: str, key: bytes | None = None, value: bytes = b"") -> None:
        ...

    async def start(self) -> None:
        ...

    async def stop(self) -> None:
        ...


class Bus:
    """Thin async wrapper around an aiokafka-compatible producer."""

    def __init__(self, producer: Any) -> None:
        self._producer = producer

    async def publish_raw(self, topic: str, key: str, payload: bytes) -> None:
        """Publish raw vendor bytes to a raw.<source> topic."""
        await self._producer.send_and_wait(topic, key=key.encode(), value=payload)

    async def publish_event(self, topic: str, key: str, event: _EventBase) -> None:
        """Serialize a canonical Event model and publish to events.normalized."""
        value = event.model_dump_json().encode()
        await self._producer.send_and_wait(topic, key=key.encode(), value=value)

    """
    Factory for a production aiokafka producer. Call .start() before use and .stop() on shutdown.

    Example:
        producer = make_producer("localhost:9092")
        await producer.start()
        bus = Bus(producer)
        ...
        await producer.stop()
    """
    from aiokafka import AIOKafkaProducer  # deferred so tests never import aiokafka

    return AIOKafkaProducer(
        bootstrap_servers=bootstrap_servers,
        value_serializer=None,  # we pre-serialize to bytes
        compression_type="lz4",
        max_batch_size=131072,
        linger_ms=10,
```

```python
"""
Okta System Log entry → canonical SSDF Event mapping.

eventType prefixes handled:
  user.session.*          → AuthEvent
  user.authentication.*   → AuthEvent
  policy.evaluate_sign_on → PolicyDecisionEvent
  user.account.report_suspicious_activity → AlertEvent
  (any other)             → None (caller skips)

Non-canonical fields are placed under ext.okta to preserve full fidelity.
"""
from __future__ import annotations

from typing import Any

from connectors.common.ssdf_events import (
    _EventBase,
    build_alert_event,
    build_auth_event,
    build_policy_decision_event,
)

# Okta severity → canonical severity mapping
_SEVERITY_MAP: dict[str, str] = {
    "DEBUG": "info",
    "INFO": "info",
    "WARN": "medium",
    "ERROR": "high",
}

# credentialType values that indicate a second factor was used
_MFA_CREDENTIAL_TYPES = frozenset(
    {
        "GOOGLE_AUTHENTICATOR",
        "SMS",
        "CALL",
        "EMAIL",
        "PUSH",
        "FIDO_WEBAUTHN",
        "OKTA_VERIFY_PUSH",
        "OKTA_VERIFY_TOTP",
        "SECURITY_KEY",
        "DUO",
        "SYMANTEC_VIP",
        "RSA_TOKEN",
    }
)


def _geo_string(geo_ctx: dict[str, Any] | None) -> str | None:
    """Collapse Okta geographicalContext into a readable string."""
    if not geo_ctx:
        return None
    parts = [geo_ctx.get("city"), geo_ctx.get("state"), geo_ctx.get("country")]
    filtered = [p for p in parts if p]
    return ", ".join(filtered) if filtered else None


def _is_mfa(auth_ctx: dict[str, Any]) -> bool:
    cred_type = auth_ctx.get("credentialType") or ""
    return cred_type in _MFA_CREDENTIAL_TYPES


def _ext_okta(entry: dict[str, Any]) -> dict[str, Any]:
    """Capture the full original entry under ext.okta for fidelity."""
    return {
        "uuid": entry.get("uuid"),
        "eventType": entry.get("eventType"),
        "displayMessage": entry.get("displayMessage"),
        "debugContext": entry.get("debugContext"),
        "securityContext": entry.get("securityContext"),
        "transaction": entry.get("transaction"),
    }


def _map_auth(
    entry: dict[str, Any], source_instance: str, tenant_id: str, severity: str
) -> _EventBase:
    actor = entry.get("actor") or {}
    outcome = entry.get("outcome") or {}
    client = entry.get("client") or {}
    auth_ctx = entry.get("authenticationContext") or {}

    return build_auth_event(
        tenant_id=tenant_id,
        source_instance=source_instance,
        ts=entry["published"],
        severity=severity,
        actor=actor.get("alternateId", actor.get("displayName", "unknown")),
        outcome=outcome.get("result", "UNKNOWN"),
        mfa=_is_mfa(auth_ctx),
        auth_method=auth_ctx.get("credentialType") or auth_ctx.get("authenticationProvider") or "UNKNOWN",
        src_ip=client.get("ipAddress", "0.0.0.0"),
        geo=_geo_string(client.get("geographicalContext")),
        risk=None,
        refs=None,
        ext_okta=_ext_okta(entry),
    )


def _map_policy(
    entry: dict[str, Any], source_instance: str, tenant_id: str, severity: str
) -> _EventBase:
    outcome = entry.get("outcome") or {}
    targets = entry.get("target") or []
    policy_target = next((t for t in targets if t.get("type") == "Policy"), None)
    policy_ref = (
        policy_target.get("alternateId") or policy_target.get("displayName") or "unknown_policy"
        if policy_target
        else "unknown_policy"
    )

    return build_policy_decision_event(
        tenant_id=tenant_id,
        source_instance=source_instance,
        ts=entry["published"],
        severity=severity,
        policy_ref=policy_ref,
        decision=outcome.get("result", "UNKNOWN"),
        reason=outcome.get("reason"),
        matched_on=None,
        refs=None,
        ext_okta=_ext_okta(entry),
    )


def _map_risk_alert(
    entry: dict[str, Any], source_instance: str, tenant_id: str, severity: str
) -> _EventBase:
    client = entry.get("client") or {}
    event_type = entry.get("eventType", "unknown")
    display_message = entry.get("displayMessage", event_type)

    return build_alert_event(
        tenant_id=tenant_id,
        source_type="okta",
        source_instance=source_instance,
        ts=entry["published"],
        severity=severity,
        rule_id=event_type,
        title=display_message,
        category="suspicious_activity",
        confidence=None,
        affected_ip=client.get("ipAddress"),
        affected_user=None,
        refs=None,
        ext_vendor=_ext_okta(entry),
        ext_vendor_key="okta",
    )


# Event type prefix → handler function
_AUTH_PREFIXES = ("user.session.", "user.authentication.")
_POLICY_TYPES = frozenset({"policy.evaluate_sign_on"})
_RISK_TYPES = frozenset({"user.account.report_suspicious_activity"})


def okta_entry_to_event(
    entry: dict[str, Any],
    *,
    source_instance: str,
    tenant_id: str,
) -> _EventBase | None:
    """
    Map one Okta System Log entry to a canonical SSDF Event.

    Returns None for eventTypes that are not in scope — the caller skips them.
    """
    event_type: str = entry.get("eventType", "")
    okta_severity: str = entry.get("severity", "INFO").upper()
    severity = _SEVERITY_MAP.get(okta_severity, "info")

    if any(event_type.startswith(prefix) for prefix in _AUTH_PREFIXES):
        return _map_auth(entry, source_instance, tenant_id, severity)

    if event_type in _POLICY_TYPES:
        return _map_policy(entry, source_instance, tenant_id, severity)

    if event_type in _RISK_TYPES:
        return _map_risk_alert(entry, source_instance, tenant_id, severity)

    return None
```

```python
"""
Wazuh indexer alert document → canonical SSDF AlertEvent mapping.

Input: the `_source` dict from a Wazuh indexer (Elasticsearch-format) alert document.
Output: AlertEventModel (canonical).

Wazuh rule.level runs 0–15. We map to severity bands:
  0-3  → low
  4-6  → medium
  7-11 → high
  12+  → critical

Confidence is derived as rule.level / 14 (clamped to [0.0, 1.0]).

Category is the first "meaningful" group (skipping generic syslog/decoder groups).
"""

from typing import Any

from connectors.common.ssdf_events import AlertEventModel, build_alert_event

# Groups that convey no domain category (skip them in favour of later groups)
_GENERIC_GROUPS = frozenset({"syslog", "sshd", "pam", "ossec", "web", "access_log", "dpkg"})


def _wazuh_level_to_severity(level: int) -> str:
    if level <= 3:
        return "low"
    if level <= 6:
        return "medium"
    if level <= 11:
        return "high"
    return "critical"


def _pick_category(groups: list[str]) -> str:
    """Return the first specific group; fall back to the raw first group."""
    for group in groups:
        if group not in _GENERIC_GROUPS:
            return group
    return groups[0] if groups else "unknown"


def _ext_wazuh(source: dict[str, Any]) -> dict[str, Any]:
    """Capture non-canonical Wazuh fields under ext.wazuh."""
    return {
        "agent": source.get("agent"),
        "manager": source.get("manager"),
        "decoder": source.get("decoder"),
        "predecoder": source.get("predecoder"),
        "full_log": source.get("full_log"),
        "location": source.get("location"),
        "cluster": source.get("cluster"),
        "rule_mitre": (source.get("rule") or {}).get("mitre"),
    }


def wazuh_alert_to_event(
    source: dict[str, Any],
    *,
    source_instance: str,
    tenant_id: str,
) -> AlertEventModel:
    """
    Map one Wazuh alert `_source` dict to a canonical AlertEvent.

    Unlike Okta mapping this always returns a value — every Wazuh alert is an AlertEvent.
    """
    rule: dict[str, Any] = source.get("rule") or {}
    data: dict[str, Any] = source.get("data") or {}
    level: int = int(rule.get("level", 0))
    groups: list[str] = rule.get("groups") or ["unknown"]

    severity = _wazuh_level_to_severity(level)

        source_type="wazuh",
```

```python
"""
Okta System Log API async client.

Env vars required:
  OKTA_DOMAIN       e.g. "myorg.okta.com"  (no https://)
  OKTA_API_TOKEN    SSWS token — pass via secret_ref; never hardcode

Pagination: cursor-based via the Link header's rel="next" URL.
The `after` parameter is the full next-page URL returned from a previous call,
or None to start from `since` (now - lookback_minutes).

Rate limits: Okta System Log /api/v1/logs is limited to 20 req/s per org.
We sleep poll_interval_s (default 30s) between pages when up-to-date.
"""
from __future__ import annotations

import os
import re
from datetime import datetime, timedelta, timezone

import httpx

_LINK_NEXT_RE = re.compile(r'<([^>]+)>;\s*rel="next"')


def _parse_next_link(link_header: str | None) -> str | None:
    """Extract the `rel="next"` URL from a Link header value, or None."""
    if not link_header:
        return None
    match = _LINK_NEXT_RE.search(link_header)
    return match.group(1) if match else None


class OktaClient:
    """
    Async Okta System Log client.

    Instantiate once and reuse across poll iterations (shares the httpx connection pool).
    """

    def __init__(
        self,
        domain: str | None = None,
        api_token: str | None = None,
        lookback_minutes: int = 5,
    ) -> None:
        self._domain = domain or os.environ["OKTA_DOMAIN"]
        self._token = api_token or os.environ["OKTA_API_TOKEN"]
        self._lookback = lookback_minutes
        self._base_url = f"https://{self._domain}"
        self._client = httpx.AsyncClient(
            base_url=self._base_url,
            headers={
                "Authorization": f"SSWS {self._token}",
                "Accept": "application/json",
                "Content-Type": "application/json",
            },
            timeout=30.0,
        )

    async def fetch_page(
        self, after: str | None
    ) -> tuple[list[dict], str | None]:
        """
        Fetch one page of System Log entries.

        If `after` is a full URL (from a previous Link header), use it directly.
        If `after` is None, start from (now - lookback_minutes).

        Returns (entries, next_cursor) where next_cursor is None when caught up.
        """
        if after and after.startswith("http"):
            # Use the full URL from the Link header directly
            response = await self._client.get(after)
        else:
            since = (
                datetime.now(timezone.utc) - timedelta(minutes=self._lookback)
            ).strftime("%Y-%m-%dT%H:%M:%S.000Z")
            params: dict[str, str] = {
                "since": since,
                "limit": "100",
                "sortOrder": "ASCENDING",
            }
            if after:
                params["after"] = after
            response = await self._client.get("/api/v1/logs", params=params)

        response.raise_for_status()
        entries: list[dict] = response.json()
        next_cursor = _parse_next_link(response.headers.get("Link"))
        return entries, next_cursor

    async def aclose(self) -> None:
        await self._client.aclose()

    async def __aenter__(self) -> "OktaClient":
        return self

    async def __aexit__(self, *_: object) -> None:
        await self.aclose()
```

```python
"""
Wazuh indexer async client (Elasticsearch REST API).

Env vars required:
  WAZUH_INDEXER_URL      e.g. "https://wazuh-indexer:9200"
  WAZUH_INDEXER_USER     e.g. "admin"
  WAZUH_INDEXER_PASSWORD password — pass via secret_ref; never hardcode
  WAZUH_INDEX_PATTERN    default "wazuh-alerts-4.x-*"

Pagination: `search_after` + a sort on `timestamp` asc + `_id` asc (tie-break).
The cursor stores the last sort values as a JSON string.

Rate limits: Wazuh indexer is self-hosted; default page size 100 docs.
"""
from __future__ import annotations

import json
import os
from datetime import datetime, timedelta, timezone
from typing import Any

import httpx

_DEFAULT_INDEX = "wazuh-alerts-4.x-*"


class WazuhClient:
    """Async Wazuh indexer client using Elasticsearch _search + search_after."""

    def __init__(
        self,
        url: str | None = None,
        username: str | None = None,
        password: str | None = None,
        index_pattern: str | None = None,
        lookback_minutes: int = 5,
        page_size: int = 100,
    ) -> None:
        self._url = (url or os.environ["WAZUH_INDEXER_URL"]).rstrip("/")
        self._user = username or os.environ["WAZUH_INDEXER_USER"]
        self._password = password or os.environ["WAZUH_INDEXER_PASSWORD"]
        self._index = index_pattern or os.environ.get("WAZUH_INDEX_PATTERN", _DEFAULT_INDEX)
        self._lookback = lookback_minutes
        self._page_size = page_size
        self._client = httpx.AsyncClient(
            auth=(self._user, self._password),
            headers={"Content-Type": "application/json"},
            verify=False,  # self-signed certs common in lab deployments
            timeout=30.0,
        )

    def _since_str(self) -> str:
        return (
            datetime.now(timezone.utc) - timedelta(minutes=self._lookback)
        ).strftime("%Y-%m-%dT%H:%M:%S.%f")[:-3] + "Z"

    async def fetch_page(
        self, search_after: list[Any] | None
    ) -> tuple[list[dict], list[Any] | None]:
        """
        Fetch one page of Wazuh alert documents.

        `search_after` is the sort-value tuple from the last hit (or None for first page).
        Returns (source_docs, next_search_after) where next_search_after is None when no more hits.
        """
        query: dict[str, Any] = {
            "size": self._page_size,
            "sort": [
                {"timestamp": {"order": "asc"}},
                {"_id": {"order": "asc"}},
            ],
            "query": {
                "range": {
                    "timestamp": {
                        "gte": self._since_str(),
                    }
                }
            },
        }
        if search_after:
            query["search_after"] = search_after

        url = f"{self._url}/{self._index}/_search"
        response = await self._client.post(url, content=json.dumps(query))
        response.raise_for_status()
        body: dict[str, Any] = response.json()

        hits: list[dict] = body.get("hits", {}).get("hits", [])
        if not hits:
            return [], None

        sources = [hit for hit in hits]  # keep full hit (mapping receives _source)
        next_sa: list[Any] | None = hits[-1].get("sort") if len(hits) == self._page_size else None
        return sources, next_sa

    async def aclose(self) -> None:
        await self._client.aclose()

    async def __aenter__(self) -> "WazuhClient":
        return self

    async def __aexit__(self, *_: object) -> None:
        await self.aclose()
```

## Task 9: Okta poller (`okta/poller.py`)

Async polling loop: fetch page → map each entry → publish raw JSON to `raw.okta` → publish canonical event (if not None) to `events.normalized` → advance cursor → sleep → repeat. Cursor is persisted after each page so restarts are idempotent.

**Files:**
- Create: `py/connectors/okta/poller.py`

- [ ] **Step 1: Implement `okta/poller.py`**

Create `py/connectors/okta/poller.py`:

```python
"""
Okta System Log async polling loop.

Usage (production):
    import asyncio
    from pathlib import Path
    from aiokafka import AIOKafkaProducer
    from connectors.common.bus import Bus, make_producer
    from connectors.common.cursor import Cursor
    from connectors.okta.client import OktaClient
    from connectors.okta.poller import OktaPoller

    async def main():
        producer = make_producer("localhost:9092")
        await producer.start()
        try:
            poller = OktaPoller(
                client=OktaClient(),
                bus=Bus(producer),
                cursor=Cursor(Path("state/okta.json"), key="okta_after"),
                tenant_id="t_main",
                source_instance="okta-main",
                poll_interval_s=30,
            )
            await poller.run_forever()
        finally:
            await producer.stop()

    asyncio.run(main())
"""
from __future__ import annotations

import asyncio
import json
import logging
from typing import Any

from connectors.common.bus import Bus
from connectors.common.cursor import Cursor
from connectors.okta.client import OktaClient
from connectors.okta.mapping import okta_entry_to_event

log = logging.getLogger(__name__)

RAW_TOPIC = "raw.okta"
NORMALIZED_TOPIC = "events.normalized"


class OktaPoller:
    """
    Polls the Okta System Log API, publishes raw + canonical events to Redpanda.

    - Raw records → `raw.okta` (full vendor JSON, key = source_instance).
    - Canonical events → `events.normalized` (key = source_instance).
    - Cursor persisted after every page to `state_file`; restarts resume safely.
    """

    def __init__(
        self,
        client: OktaClient,
        bus: Bus,
        cursor: Cursor,
        tenant_id: str,
        source_instance: str,
        poll_interval_s: float = 30.0,
    ) -> None:
        self._client = client
        self._bus = bus
        self._cursor = cursor
        self._tenant_id = tenant_id
        self._source_instance = source_instance
        self._poll_interval = poll_interval_s

    async def poll_once(self) -> int:
        """
        Fetch and publish one page of Okta log entries.

        Returns the number of raw records published.
        """
        after = self._cursor.get()
        entries, next_cursor = await self._client.fetch_page(after)

        published = 0
        for entry in entries:
            raw_bytes = json.dumps(entry, separators=(",", ":")).encode()
            await self._bus.publish_raw(RAW_TOPIC, key=self._source_instance, payload=raw_bytes)

            event = okta_entry_to_event(
                entry,
                source_instance=self._source_instance,
                tenant_id=self._tenant_id,
            )
            if event is not None:
                await self._bus.publish_event(NORMALIZED_TOPIC, key=self._source_instance, event=event)

            published += 1

        if next_cursor:
            self._cursor.set(next_cursor)
        elif entries:
            # No next page: store the uuid of the last entry so the next poll
            # starts just after it (Okta `after` param accepts a log event uuid).
            last_uuid = entries[-1].get("uuid")
            if last_uuid:
                self._cursor.set(last_uuid)

        log.info(
            "okta poll_once: published=%d next_cursor=%s",
            published,
            next_cursor or "(caught up)",
        )
        return published

    async def run_forever(self) -> None:
        """Run the polling loop indefinitely. Call from asyncio.run()."""
        log.info(
            "OktaPoller starting: source_instance=%s poll_interval=%.0fs",
            self._source_instance,
            self._poll_interval,
        )
        while True:
            try:
                count = await self.poll_once()
                if count == 0:
                    # Caught up — wait full interval before next poll
                    await asyncio.sleep(self._poll_interval)
                else:
                    # More pages may be available — short sleep then continue
                    await asyncio.sleep(0.5)
            except asyncio.CancelledError:
                log.info("OktaPoller cancelled — shutting down")
                break
            except Exception as exc:
                log.error("OktaPoller error: %s — retrying in %ss", exc, self._poll_interval)
                await asyncio.sleep(self._poll_interval)
```
